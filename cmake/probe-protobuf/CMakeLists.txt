set(CMAKE_FIND_PACKAGE_TARGETS_GLOBAL FALSE)
find_package(Protobuf QUIET CONFIG)
if(NOT TARGET protobuf::libprotobuf OR NOT TARGET protobuf::protoc)
  find_package(Protobuf QUIET MODULE)
endif()

if(TARGET protobuf::libprotobuf AND TARGET protobuf::protoc)
  get_target_property(protoc_executable protobuf::protoc IMPORTED_LOCATION)
  if(NOT protoc_executable)
    get_target_property(protoc_executable protobuf::protoc IMPORTED_LOCATION_RELEASE)
  endif()
  execute_process(COMMAND "${protoc_executable}" --version
    RESULT_VARIABLE protoc_result OUTPUT_VARIABLE protoc_output ERROR_QUIET)
  string(REGEX MATCH "[0-9]+\\.[0-9]+(\\.[0-9]+)?" protoc_version "${protoc_output}")
  get_target_property(include_dirs protobuf::libprotobuf INTERFACE_INCLUDE_DIRECTORIES)
  set(compatible FALSE)
  foreach(include_dir IN LISTS include_dirs)
    foreach(version_header IN ITEMS runtime_version.h stubs/common.h)
      set(header "${include_dir}/google/protobuf/${version_header}")
      if(EXISTS "${header}" AND EXISTS "${include_dir}/google/protobuf/timestamp.proto")
        file(STRINGS "${header}" version_line
          REGEX "^#define (GOOGLE_)?PROTOBUF_VERSION [0-9]+")
        if(version_line MATCHES "PROTOBUF_VERSION ([0-9]+)")
          set(version_number "${CMAKE_MATCH_1}")
          math(EXPR version_major "${version_number} / 1000000")
          math(EXPR version_minor "(${version_number} / 1000) % 1000")
          math(EXPR version_patch "${version_number} % 1000")
          if(protoc_result EQUAL 0 AND
              (protoc_version STREQUAL "${version_major}.${version_minor}.${version_patch}" OR
               protoc_version STREQUAL "${version_minor}.${version_patch}"))
            set(compatible TRUE)
          endif()
        endif()
      endif()
    endforeach()
  endforeach()
  if(compatible)
    get_property(imported_targets DIRECTORY PROPERTY IMPORTED_TARGETS)
    foreach(imported_target IN LISTS imported_targets)
      set_property(TARGET ${imported_target} PROPERTY IMPORTED_GLOBAL TRUE)
    endforeach()
    set(LOKIVERSE_SYSTEM_PROTOBUF_FOUND TRUE PARENT_SCOPE)
  endif()
endif()
